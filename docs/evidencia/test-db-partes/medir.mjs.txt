// node medir.mjs <dir-logs> <raiz-repo> [<raiz-extra>...] : duración de cada archivo de test:db en los logs del CI.
// Toma las pruebas de primer nivel («test:db: ✔ título (ms)»), busca el título en los archivos de la lista y suma por
// archivo en cada corrida; luego da el máximo y la mediana por archivo.
import { readdirSync, readFileSync, existsSync } from "node:fs";
import { join } from "node:path";

const [, , dir, ...raices] = process.argv;
const lista = (raiz) => readFileSync(join(raiz, "scripts/ci/pruebas-postgres.sh"), "utf8").match(/tests\/[^\s\\]+\.test\.ts/g);
const archivos = new Map();
for (const raiz of raices) for (const a of lista(raiz)) if (!archivos.has(a) && existsSync(join(raiz, a))) archivos.set(a, readFileSync(join(raiz, a), "utf8"));
const deTitulo = (titulo) => {
  const hallados = [...archivos].filter(([, fuente]) => fuente.includes(JSON.stringify(titulo).slice(1, -1)) || fuente.includes(titulo));
  return hallados.length === 1 ? hallados[0][0] : hallados.length === 0 ? "?" : `ambiguo:${hallados.map(([a]) => a).join(",")}`;
};
const porArchivo = new Map();
const sinArchivo = new Set();
for (const f of readdirSync(dir).filter((x) => x.endsWith(".txt"))) {
  const texto = readFileSync(join(dir, f), "utf8");
  const suma = new Map();
  for (const m of texto.matchAll(/"message": "test:db: (?:✔|✖) (.*) \(([0-9.]+)ms\)"/g)) {
    const titulo = JSON.parse(`"${m[1]}"`);
    const a = deTitulo(titulo);
    if (a.startsWith("?") || a.startsWith("ambiguo")) sinArchivo.add(`${a} ${titulo}`);
    suma.set(a, (suma.get(a) ?? 0) + Number(m[2]) / 1000);
  }
  if (suma.size === 0) continue;
  for (const [a, s] of suma) porArchivo.set(a, [...(porArchivo.get(a) ?? []), s]);
}
const filas = [...porArchivo].map(([a, v]) => {
  const o = [...v].sort((x, y) => x - y);
  return { archivo: a, corridas: v.length, max: o.at(-1), mediana: o[Math.floor(o.length / 2)] };
}).sort((x, y) => y.max - x.max);
for (const f of filas) console.log(`${f.max.toFixed(1).padStart(7)} s máx · ${f.mediana.toFixed(1).padStart(7)} s mediana · ${f.corridas} corridas · ${f.archivo}`);
console.log(`total máx ${filas.reduce((s, f) => s + f.max, 0).toFixed(0)} s · mediana ${filas.reduce((s, f) => s + f.mediana, 0).toFixed(0)} s`);
if (sinArchivo.size) console.log("SIN ARCHIVO:\n" + [...sinArchivo].join("\n"));
for (const raiz of raices) for (const a of lista(raiz)) if (!porArchivo.has(a)) console.log("sin medición:", a);
