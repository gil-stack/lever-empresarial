// Línea de tiempo del rojo «riesgo, sin JavaScript»: desde el clic hasta 3 s después de load, sin JavaScript en la
// página. Por CDP: caja del documento (desplazamiento), caja del campo y del mensaje, y la fuente que de verdad pinta
// un título en Sora (CSS.getPlatformFontsForNode). Marca domcontentloaded y load con los eventos de Playwright.
// ANCHO=390|1440 VECES=n SWIFT=1 (como R1). Imprime una línea JSON por corrida.
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";
import { chromium } from "../../../../node_modules/playwright/index.mjs";

const E = JSON.parse(readFileSync(process.env.ESTADO, "utf8"));
const BASE = process.env.BASE;
const sql = (q) => execFileSync("psql", [E.DUENO_URL, "-X", "-q", "-tA", "-v", "ON_ERROR_STOP=1", "-c", q], { encoding: "utf8" }).trim();
const brisa = sql("select id from companies where rfc = 'BRR180303CD6'");
const ancho = Number(process.env.ANCHO ?? 1440);
const veces = Number(process.env.VECES ?? 1);
const movil = ancho < 600;

for (let corrida = 1; corrida <= veces; corrida++) {
  sql("update sessions set ultimo_uso = now() where expires_at > now()");
  const nav = await chromium.launch(process.env.SWIFT === "0" ? {} : { args: ["--use-angle=swiftshader", "--enable-unsafe-swiftshader"] });
  const ctx = await nav.newContext({ viewport: { width: ancho, height: movil ? 844 : 900 }, isMobile: movil, hasTouch: movil, deviceScaleFactor: 1,
    javaScriptEnabled: false, reducedMotion: "reduce", locale: "es-MX", timezoneId: "America/Mexico_City" });
  const host = new URL(BASE).hostname;
  await ctx.addCookies([
    { name: "tema", value: "oscuro", domain: host, path: "/" },
    { name: "__Host-sesion", value: E.SESIONES.celia, domain: host, path: "/", httpOnly: true, secure: true, sameSite: "Lax" },
  ]);
  const p = await ctx.newPage();
  const cdp = await ctx.newCDPSession(p);
  await cdp.send("DOM.enable");
  await cdp.send("CSS.enable");
  await p.goto(`${BASE}/operacion/clientes/${brisa}/expediente`, { waitUntil: "networkidle" });
  await p.locator("#riesgo-actividad").selectOption("media");
  await p.locator("#riesgo-zona").selectOption("alta");
  await p.route("**/operacion/**", async (route) => {
    const cuerpo = route.request().postDataBuffer();
    if (route.request().method() !== "POST" || !cuerpo) return route.continue();
    return route.continue({ postData: Buffer.from(cuerpo.toString("latin1").replace(/(name="[^"]*actividad"\r\n\r\n)media/, "$1MEDIA"), "latin1") });
  });
  const t0 = Date.now();
  const marcas = {};
  p.on("domcontentloaded", () => { marcas.dcl ??= Date.now() - t0; });
  p.on("load", () => { marcas.load ??= Date.now() - t0; });
  const caja = async (sel) => {
    try {
      const { root } = await cdp.send("DOM.getDocument", { depth: 0 });
      const { nodeId } = await cdp.send("DOM.querySelector", { nodeId: root.nodeId, selector: sel });
      if (!nodeId) return null;
      const { model } = await cdp.send("DOM.getBoxModel", { nodeId });
      return { y: Math.round(model.border[1]), nodeId };
    } catch { return null; }
  };
  const fuenteDelTitulo = async () => {
    try {
      const t = await caja("main h1");
      if (!t) return null;
      const { fonts } = await cdp.send("CSS.getPlatformFontsForNode", { nodeId: t.nodeId });
      return fonts.map((f) => f.familyName).join("+");
    } catch { return null; }
  };
  const linea = [];
  await p.getByRole("button", { name: "Registrar revisión periódica" }).click({ noWaitAfter: true });
  let fin = Infinity;
  while (Date.now() - t0 < Math.min(fin, 15000)) {
    const html = await caja("html");
    const campo = await caja("#riesgo-actividad:focus");
    const mensaje = await caja('[id="aviso-riesgo-campo"]');
    const fila = { scrollY: html ? -html.y : null, campo: campo?.y ?? null, mensaje: mensaje?.y ?? null, fuente: await fuenteDelTitulo() };
    const ultimo = linea.at(-1);
    if (!ultimo || JSON.stringify(ultimo.f) !== JSON.stringify(fila)) linea.push({ ms: Date.now() - t0, f: fila });
    if (marcas.load !== undefined && fin === Infinity) fin = marcas.load + 3000;
  }
  const barras = [];
  for (const sel of [".barra-movil", ".barra-superior"]) {
    const b = await p.locator(sel).first().boundingBox().catch(() => null);
    if (b && b.y <= 1 && b.height > 0) barras.push(Math.round(b.y + b.height));
  }
  const tapa = Math.max(0, ...barras);
  const ultimo = linea.at(-1)?.f ?? {};
  const ok = ultimo.mensaje !== null && ultimo.mensaje >= tapa && ultimo.campo !== null;
  console.log(JSON.stringify({ corrida, ancho, ok, tapa, marcas, final: ultimo, linea }));
  await nav.close();
}
