// Reparto REAL: `playwright test --list --shard=n/K` (orden de definición, el que usa Playwright), con el tiempo máximo
// observado de cada prueba en el CI (fijo 90 s por parte). Cuenta las pruebas normales que corren después de una con GPU
// emulada en la misma parte.
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";
const K = Number(process.argv[2]), FIJO = 90;
const T = JSON.parse(readFileSync(process.argv[3], "utf8"));
const tiempo = (k) => T[k] ?? T[k.replace(/tests\/e2e\/[\w-]+\.spec\.ts › (fondo animado)/, "tests/e2e/fondo-pruebas.ts › $1")] ?? 5;
let peor = 0, total = 0;
for (let n = 1; n <= K; n++) {
  const lineas = execFileSync("./node_modules/.bin/playwright", ["test", "--list", `--shard=${n}/${K}`], { encoding: "utf8" }).split("\n").filter((l) => /\.ts:\d+:\d+ › /.test(l));
  let s = 0, vista = false, despues = 0; const por = {};
  for (const l of lineas) {
    const archivo = /([\w-]+\.(?:spec\.)?ts):\d+:\d+ › /.exec(l)[1];
    const k = `tests/e2e/${archivo} › ${l.split(/\.ts:\d+:\d+ › /)[1].trim()}`;
    const t = tiempo(k); s += t; por[archivo] = (por[archivo] ?? 0) + t;
    if (/con GPU rápida/.test(k)) vista = true; else if (vista) despues++;
  }
  total += despues; peor = Math.max(peor, s + FIJO);
  console.log(`parte ${n}/${K}: ${String(lineas.length).padStart(3)} pruebas · ${s.toFixed(0).padStart(4)} s de pruebas · ~${(s + FIJO).toFixed(0)} s · normales después de GPU: ${despues} · ${Object.entries(por).map(([a, v]) => `${a.replace(".spec.ts", "")} ${v.toFixed(0)}`).join(", ")}`);
}
console.log(`peor parte ~${peor.toFixed(0)} s · normales después de una con GPU: ${total}`);
