const medir = () => {
  const leer = (css) => { const n = (css.match(/-?[\d.]+%?/g) ?? []).map((x) => (x.endsWith("%") ? parseFloat(x) / 100 : parseFloat(x))); return css.startsWith("color(srgb") ? [n[0] * 255, n[1] * 255, n[2] * 255, n[3] ?? 1] : [n[0] ?? 0, n[1] ?? 0, n[2] ?? 0, n[3] ?? 1]; };
  const sobre = ([r, g, b, a], [R, G, B]) => [r * a + R * (1 - a), g * a + G * (1 - a), b * a + B * (1 - a), 1];
  const luz = ([r, g, b]) => { const c = (v) => ((v /= 255) <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4); return 0.2126 * c(r) + 0.7152 * c(g) + 0.0722 * c(b); };
  const razon = (x, y) => { const [a, b] = [luz(x), luz(y)].sort((p, q) => q - p); return (a + 0.05) / (b + 0.05); };
  const fallas = []; let textos = 0, minimo = 99;
  for (const barra of document.querySelectorAll(".barra-movil, .barra-superior")) {
    if (!barra.getClientRects().length) continue;
    for (const el of barra.querySelectorAll("*")) {
      const texto = [...el.childNodes].filter((n) => n.nodeType === 3).map((n) => n.textContent).join("").trim();
      const caja = el.getBoundingClientRect(), estilo = getComputedStyle(el);
      if (!texto || !caja.width || !caja.height || estilo.visibility === "hidden" || el.closest("[hidden], .solo-lector")) continue;
      const capas = []; for (let x = el; x; x = x === barra ? null : x.parentElement) capas.unshift(leer(getComputedStyle(x).backgroundColor));
      const grande = parseFloat(estilo.fontSize) >= 24 || (parseFloat(estilo.fontSize) >= 18.66 && Number(estilo.fontWeight) >= 700);
      textos++;
      for (const [nombre, debajo] of [["negro", [0, 0, 0, 1]], ["blanco", [255, 255, 255, 1]]]) {
        const fondo = capas.reduce((acum, capa) => sobre(capa, acum), debajo); const r = razon(sobre(leer(estilo.color), fondo), fondo);
        minimo = Math.min(minimo, r); if (r < (grande ? 3 : 4.5)) fallas.push(`«${texto.slice(0, 30)}» ${nombre} ${r.toFixed(2)}`);
      }
    }
  }
  return { textos, minimo: minimo.toFixed(2), fallas, desplazada: document.querySelector(".barra-superior")?.hasAttribute("data-desplazada") ?? null };
};
