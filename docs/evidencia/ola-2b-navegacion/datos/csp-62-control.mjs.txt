// Control de la sonda: un <style> en línea sin nonce tiene que producir aviso en la consola y evento.
import { createRequire } from "node:module";
const W = "/Users/gilbertovillarreal/dev/plataforma-administrativa/.worktrees/ola-2b-navegacion";
const { chromium, webkit } = createRequire(`${W}/package.json`)("playwright");
const POLITICA = /content security policy|refused to (load|apply|execute)|blocked by csp|violates the following/i;
for (const motor of ["chromium", "webkit"]) {
  const nav = await (motor === "webkit" ? webkit : chromium).launch({ headless: true });
  const ctx = await nav.newContext({ ignoreHTTPSErrors: true });
  await ctx.addInitScript(() => { window.__csp = []; document.addEventListener("securitypolicyviolation", (e) => window.__csp.push(e.violatedDirective)); });
  const p = await ctx.newPage(); const avisos = [];
  p.on("console", (m) => { if (POLITICA.test(m.text())) avisos.push(m.text().slice(0, 100)); });
  await p.goto("https://localhost:3152/entrar", { waitUntil: "load" });
  await p.evaluate(() => { const s = document.createElement("style"); s.textContent = "body { outline: 3px solid red }"; document.head.append(s); window.__inyectado = getComputedStyle(document.body).outlineWidth; });
  await p.waitForTimeout(500);
  console.log(motor, "avisos:", avisos.length, avisos[0] ?? "", "| eventos:", (await p.evaluate(() => window.__csp)).join(","), "| outline aplicado:", await p.evaluate(() => getComputedStyle(document.body).outlineWidth));
  await nav.close();
}
