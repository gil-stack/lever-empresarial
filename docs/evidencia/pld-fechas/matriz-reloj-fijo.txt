## a-la-hora.sh
#!/bin/bash
# Uso: a-la-hora.sh "<UTC YYYY-MM-DD HH:MM:SS>" <salida.log>
# Postgres 5553 y Node con libfaketime arrancando en ese instante (el reloj avanza).
set -u
S=<scratchpad>
M=<scratchpad>
RAIZ=${RAIZ:-<raíz del repo>}
PGB=/opt/homebrew/opt/postgresql@17/bin
FT=(env TZ=UTC DYLD_FORCE_FLAT_NAMESPACE=1 DYLD_INSERT_LIBRARIES=/opt/homebrew/lib/faketime/libfaketime.1.dylib FAKETIME="@$1" FAKETIME_DONT_RESET=1)
"${FT[@]}" LC_ALL=C $PGB/postgres -D $S/pg5553 -p 5553 -c unix_socket_directories='' >> $S/pg5553.log 2>&1 &
pg=$!
for i in $(seq 1 50); do $PGB/pg_isready -q -h 127.0.0.1 -p 5553 && break; sleep 0.2; done
echo "postgres pid=$pg now()=$($PGB/psql -h 127.0.0.1 -p 5553 -U postgres -Atc "select now() at time zone 'America/Mexico_City'")" > "$2"
cd "$RAIZ"
"${FT[@]}" PATH="$M/pf/bin:$PATH" AISLAMIENTO_DATABASE_URL=postgres://postgres@127.0.0.1:5553/postgres node -e 'console.log("node CDMX", new Date().toLocaleString("sv-SE",{timeZone:"America/Mexico_City"}))' >> "$2" 2>&1
"${FT[@]}" PATH="$M/pf/bin:$PATH" AISLAMIENTO_DATABASE_URL=postgres://postgres@127.0.0.1:5553/postgres /opt/homebrew/bin/node node_modules/tsx/dist/cli.mjs --test --test-concurrency=1 ${PRUEBAS:-tests/pld/pld-postgres.test.ts} >> "$2" 2>&1
echo "salida=$?" >> "$2"
kill -INT $pg; wait $pg 2>/dev/null
