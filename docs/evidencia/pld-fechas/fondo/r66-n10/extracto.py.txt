# extracto.py <log crudo> <salida> [--sin-errores]: sin ANSI; líneas de lista, resumen y, de cada ✘, título y error.
import re, sys
crudo = re.sub(r"\x1b\[[0-9;]*[A-Za-z]", "", open(sys.argv[1], encoding="utf8", errors="replace").read()).splitlines()
lista = re.compile(r"^\s+[✓✘-]\s+\d+ tests/")
resumen = re.compile(r"^\s+\d+ (passed|failed|flaky|skipped|did not run|interrupted)\b|^\[e2e\] recorridos terminados|^salida=")
cabeza = re.compile(r"^\s+\d+\) tests/")
corte = re.compile(r"^\s+(Call Log:|at |attachment #|Error Context:)|^\s+>?\s*\d+ \|")
o, bloque, n = [], False, 0
for l in crudo:
    if lista.match(l) or resumen.match(l):
        o.append(l); bloque = False; continue
    if "--sin-errores" in sys.argv: continue
    if cabeza.match(l):
        o.append(""); o.append(l.rstrip()); bloque, n = True, 0; continue
    if bloque:
        if corte.match(l): bloque = False; continue
        if l.strip() and n < 14:
            o.append(l[:600]); n += 1
open(sys.argv[2], "w", encoding="utf8").write("\n".join(o) + "\n")
