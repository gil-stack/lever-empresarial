assert s.count(a)==1, p

  if (!remoto && process.env.E2E_SWIFT) return playwright.chromium.launch({ ...launchOptions, args: [...(launchOptions.args ?? []), "--use-gl=angle", "--use-angle=swiftshader", "--enable-unsafe-swiftshader"] });
  if (!remoto) return playwright.chromium.launch(launchOptions);''')
open(p,"w").write(s)
