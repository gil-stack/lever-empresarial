## a-la-hora.sh
#!/bin/bash
# Uso: a-la-hora.sh "<UTC YYYY-MM-DD HH:MM:SS>" <salida> <TZ del proceso> <timezone de Postgres> [pruebas...]
set -u
S=<scratchpad>
INSTANTE=$1; SALIDA=$2; export TZ=$3; PGZ=$4; shift 4
export DYLD_FORCE_FLAT_NAMESPACE=1 DYLD_INSERT_LIBRARIES=/opt/homebrew/lib/faketime/libfaketime.1.dylib FAKETIME="@$INSTANTE" FAKETIME_DONT_RESET=1
PGB=/opt/homebrew/opt/postgresql@17/bin
$PGB/postgres -D $S/pg5553 -p 5553 -c unix_socket_directories='' -c listen_addresses=127.0.0.1 -c timezone=$PGZ >> $S/pg5553.log 2>&1 &
pg=$!
for i in $(seq 1 50); do $PGB/pg_isready -q -h 127.0.0.1 -p 5553 && break; sleep 0.2; done
{ echo "TZ=$TZ pg.timezone=$PGZ"; echo "pg now() CDMX: $($PGB/psql -h 127.0.0.1 -p 5553 -U postgres -Atc "select now() at time zone 'America/Mexico_City'")"; /opt/homebrew/bin/node -e 'console.log("node CDMX", new Date().toLocaleString("sv-SE",{timeZone:"America/Mexico_City"}), "· local", new Date().toString().slice(0,33))'; } > "$SALIDA"
unset DATABASE_URL
AISLAMIENTO_DATABASE_URL=postgres://postgres@127.0.0.1:5553/postgres /opt/homebrew/bin/node node_modules/tsx/dist/cli.mjs --test --test-concurrency=1 "$@" >> "$SALIDA" 2>&1
echo "salida=$?" >> "$SALIDA"
kill -INT $pg; wait $pg 2>/dev/null
